cmake_minimum_required(VERSION 3.10)
project(go2_place_mtc)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_task_constructor_core REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(go2_place_mtc src/place.cpp)
ament_target_dependencies(go2_place_mtc rclcpp moveit_task_constructor_core moveit_core moveit_ros_planning moveit_msgs geometry_msgs)
target_link_libraries(go2_place_mtc nlohmann_json::nlohmann_json)
target_compile_features(go2_place_mtc PRIVATE cxx_std_17)
install(TARGETS go2_place_mtc DESTINATION lib/${PROJECT_NAME})
ament_package()
